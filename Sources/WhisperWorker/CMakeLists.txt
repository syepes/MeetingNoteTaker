cmake_minimum_required(VERSION 3.21)
project(MeetingWhisperWorker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper)
add_executable(whisper-worker main.cpp)
target_include_directories(whisper-worker PRIVATE ${WHISPER_SOURCE}/examples)
target_link_libraries(whisper-worker PRIVATE common whisper)
